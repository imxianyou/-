        function startSurvey() {

            document.getElementById('view-intro').classList.add('hidden');

        // 2. 질문 렌더링

        function renderQuestion() {

            const q = questions[currentStep];

            const progress = ((currentStep + 1) / questions.length) * 100;

            

            document.getElementById('step-info').innerText = (currentStep + 1).toString().padStart(2, '0');

            document.getElementById('progress-val').innerText = `STEP ${currentStep + 1}/${questions.length}`;

            document.getElementById('progress-fill').style.width = progress + '%';

            document.getElementById('question-text').innerText = q.q;


            const container = document.getElementById('options-container');

            container.innerHTML = '';


            q.options.forEach(opt => {

                const btn = document.createElement('button');

                btn.className = 'option-btn fade-in';

                btn.innerHTML = `

                    </div>

                    <span>${opt.t}</span>

                `;

                btn.onclick = () => selectOption(q.key, opt.v);

                container.appendChild(btn);

            });

        }


        // 3. 답변 선택

        function selectOption(key, val) {

            answers[key] = val;

            if (currentStep < questions.length - 1) {

                currentStep++;

                renderQuestion();

            } else {

                showResult();

            }

        }


        // 4. 결과 출력

        function showResult() {

            const res = database.find(r => r.cond(answers)) || database[database.length-1];

            document.getElementById('res-premium-name').innerText = res.premium;

            document.getElementById('res-budget-name').innerText = res.budget;

            document.getElementById('res-description').innerText = res.desc;

            document.getElementById('res-style-tag').innerText = res.style;


            document.getElementById('view-survey').classList.add('hidden');

            document.getElementById('view-result').classList.remove('hidden');

        }
